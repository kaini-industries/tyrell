cmake_minimum_required(VERSION 3.25)
project(tyrell_status VERSION 0.1.1 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr" CACHE PATH "Install prefix" FORCE)
endif()

set(LVGL_SOURCE_DIR "" CACHE PATH "Verified LVGL 9.5.0 source directory")
set(TYRELL_BUILD_REVISION "dev" CACHE STRING "Source revision displayed by the app")
option(TYRELL_ENABLE_SDL "Enable the optional SDL desktop simulator" OFF)
option(TYRELL_ENABLE_SHELL_LINK "Enable the experimental native Gh0st shell-link pilot" OFF)
include(CTest)
if(TYRELL_ENABLE_SHELL_LINK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(TYRELL_APP_COMMON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tyrell-app-common" CACHE PATH "Shared Tyrell app primitives")
  option(TYRELL_APP_COMMON_BUILD_SHELL_LINK_TESTS "Build shared shell transport tests" ${BUILD_TESTING})
  add_subdirectory("${TYRELL_APP_COMMON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/app-common")
endif()
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the verified LVGL source from locks/build-inputs.json; configuration never downloads dependencies.")
endif()

# Disable external image/font dependencies and library examples. LVGL is linked
# statically into this package; EXCLUDE_FROM_ALL excludes its install rules.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC TYRELL_ENABLE_SDL=$<BOOL:${TYRELL_ENABLE_SDL}>)
if(TYRELL_ENABLE_SDL)
  find_package(SDL2 CONFIG REQUIRED)
  target_link_libraries(lvgl PUBLIC SDL2::SDL2)
endif()
if(UNIX)
  target_link_libraries(lvgl PUBLIC m)
endif()

add_library(status_model STATIC src/status_model.cpp)
target_include_directories(status_model PUBLIC src)
target_compile_options(status_model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tyrell-status src/main.cpp)
target_link_libraries(tyrell-status PRIVATE lvgl::lvgl status_model)
target_compile_definitions(tyrell-status PRIVATE
  TYRELL_VERSION="${PROJECT_VERSION}" TYRELL_REVISION="${TYRELL_BUILD_REVISION}"
  TYRELL_ENABLE_SHELL_LINK=$<BOOL:${TYRELL_ENABLE_SHELL_LINK}>)
if(TYRELL_ENABLE_SHELL_LINK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(tyrell-status PRIVATE Tyrell::ShellLink)
endif()
target_compile_options(tyrell-status PRIVATE -Wall -Wextra -Wpedantic)

# The builder may run on newer ARM hardware. CM0 is a Cortex-A53 (ARMv8-A).
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  foreach(target lvgl status_model tyrell-status)
    target_compile_options(${target} PRIVATE -march=armv8-a -mtune=cortex-a53)
  endforeach()
endif()

if(BUILD_TESTING)
  add_executable(status-model-test tests/status_model_test.cpp)
  target_link_libraries(status-model-test PRIVATE status_model)
  target_compile_options(status-model-test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME status-model COMMAND status-model-test)
  add_executable(status-input-edges-test tests/input_edges_test.cpp)
  target_include_directories(status-input-edges-test PRIVATE src)
  target_compile_options(status-input-edges-test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME status-input-edges COMMAND status-input-edges-test)
  add_test(NAME cli-version COMMAND tyrell-status --version)
  add_test(NAME cli-help COMMAND tyrell-status --help)
  add_test(NAME headless-render COMMAND tyrell-status
    --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/status-test.ppm" --frames 2)
  if(TYRELL_ENABLE_SHELL_LINK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(status-shell-lifecycle-test tests/shell_lifecycle_test.cpp)
    target_include_directories(status-shell-lifecycle-test PRIVATE src)
    target_link_libraries(status-shell-lifecycle-test PRIVATE Tyrell::ShellLink)
    target_compile_options(status-shell-lifecycle-test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME status-shell-lifecycle COMMAND status-shell-lifecycle-test)
    add_executable(status-shell-gui-test tests/shell_gui_test.cpp)
    target_link_libraries(status-shell-gui-test PRIVATE lvgl::lvgl status_model Tyrell::ShellLink)
    target_compile_definitions(status-shell-gui-test PRIVATE TYRELL_ENABLE_SHELL_LINK=1
      TYRELL_VERSION="${PROJECT_VERSION}" TYRELL_REVISION="test")
    target_compile_options(status-shell-gui-test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME status-shell-gui COMMAND status-shell-gui-test)
    set_tests_properties(status-shell-lifecycle status-shell-gui PROPERTIES TIMEOUT 10)
  endif()
endif()

install(TARGETS tyrell-status RUNTIME DESTINATION bin)
install(FILES assets/tyrell-status.desktop DESTINATION share/APPLaunch/applications)
install(FILES assets/tyrell-status-standard.desktop DESTINATION share/applications RENAME tyrell-status.desktop)
install(FILES assets/tyrell-status.png DESTINATION share/APPLaunch/share/images)
install(FILES assets/tyrell-status.png DESTINATION share/icons/hicolor/80x80/apps)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION share/doc/tyrell-status)
if(TYRELL_ENABLE_SHELL_LINK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  install(FILES "${TYRELL_APP_COMMON_SOURCE_DIR}/LICENSE" DESTINATION share/doc/tyrell-status RENAME app-common-LICENSE)
endif()
install(FILES "${LVGL_SOURCE_DIR}/LICENCE.txt" DESTINATION share/doc/tyrell-status RENAME lvgl-LICENCE.txt)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  execute_process(COMMAND dpkg --print-architecture OUTPUT_VARIABLE TYRELL_DEBIAN_ARCH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT TYRELL_DEBIAN_ARCH STREQUAL "arm64")
    message(FATAL_ERROR "Device package must be built natively on Debian ARM64; got ${TYRELL_DEBIAN_ARCH}")
  endif()
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME tyrell-status)
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_CONTACT "Tyrell contributors <tyrell@localhost>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Read-only system status for the M5Stack CardputerZero")
  set(CPACK_PACKAGE_DESCRIPTION "Small LVGL status application showing uptime, memory, battery and network state. Includes headless JSON output and a 320x170 preview renderer.")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
  set(CPACK_DEBIAN_PACKAGE_NAME tyrell-status)
  set(CPACK_DEBIAN_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_DEBIAN_PACKAGE_RELEASE 1)
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
  set(CPACK_DEBIAN_PACKAGE_SECTION utils)
  set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
  set(CPACK_PACKAGE_FILE_NAME "tyrell-status_${PROJECT_VERSION}-1_arm64")
  include(CPack)
endif()
