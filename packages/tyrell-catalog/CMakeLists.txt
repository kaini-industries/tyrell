cmake_minimum_required(VERSION 3.25)
project(tyrell_catalog VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX /usr CACHE PATH "Install prefix" FORCE)
endif()
include(CTest)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(LVGL_SOURCE_DIR "" CACHE PATH "Verified locked LVGL source")
set(TYRELL_BUILD_REVISION "dev" CACHE STRING "Source revision displayed by Catalog")
set(TYRELL_APP_COMMON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tyrell-app-common" CACHE PATH "Shared command metadata")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the verified source from locks/build-inputs.json; configuration never downloads dependencies")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
# Catalog artwork deliberately excludes profiles/text chunks; the decoder also
# omits their secondary inflation paths, including in explicit preview mode.
target_compile_definitions(lvgl PRIVATE LODEPNG_NO_COMPILE_ANCILLARY_CHUNKS)
if(UNIX)
  target_link_libraries(lvgl PUBLIC m)
endif()
add_library(catalog_model STATIC src/catalog_model.cpp src/artwork.cpp)
target_include_directories(catalog_model PUBLIC src "${TYRELL_APP_COMMON_SOURCE_DIR}/include")
target_link_libraries(catalog_model PUBLIC OpenSSL::Crypto)
target_compile_options(catalog_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(tyrell-catalog src/main.cpp src/backend_process.cpp)
target_link_libraries(tyrell-catalog PRIVATE lvgl::lvgl catalog_model)
target_compile_definitions(tyrell-catalog PRIVATE TYRELL_VERSION="${PROJECT_VERSION}" TYRELL_REVISION="${TYRELL_BUILD_REVISION}")
target_compile_options(tyrell-catalog PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  foreach(target lvgl catalog_model tyrell-catalog)
    target_compile_options(${target} PRIVATE -march=armv8-a -mtune=cortex-a53)
  endforeach()
endif()
if(BUILD_TESTING)
  add_executable(catalog-model-test tests/model_test.cpp)
  target_link_libraries(catalog-model-test PRIVATE catalog_model)
  target_compile_options(catalog-model-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME catalog-model COMMAND catalog-model-test)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/assets/tyrell-catalog.png" CATALOG_ICON_SHA)
  add_executable(catalog-artwork-test tests/artwork_test.cpp)
  target_link_libraries(catalog-artwork-test PRIVATE catalog_model)
  target_compile_definitions(catalog-artwork-test PRIVATE
    CATALOG_TEST_ICON="${CMAKE_CURRENT_SOURCE_DIR}/assets/tyrell-catalog.png" CATALOG_TEST_ICON_SHA="${CATALOG_ICON_SHA}")
  target_compile_options(catalog-artwork-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME catalog-artwork COMMAND catalog-artwork-test)
  add_executable(catalog-backend-process-test tests/process_test.cpp src/backend_process.cpp)
  target_include_directories(catalog-backend-process-test PRIVATE src)
  target_compile_definitions(catalog-backend-process-test PRIVATE TYRELL_CATALOG_PROCESS_TEST=1)
  target_compile_options(catalog-backend-process-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME catalog-backend-process COMMAND catalog-backend-process-test)
  set_tests_properties(catalog-backend-process PROPERTIES TIMEOUT 15)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(catalog-entry-guard-test tests/entry_guard_test.cpp src/backend_process.cpp)
    target_link_libraries(catalog-entry-guard-test PRIVATE lvgl::lvgl catalog_model)
    target_compile_definitions(catalog-entry-guard-test PRIVATE TYRELL_VERSION="${PROJECT_VERSION}" TYRELL_REVISION="test")
    target_compile_options(catalog-entry-guard-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(catalog-entry-guard-test PRIVATE -Wl,--wrap=fork)
    add_test(NAME catalog-entry-guard COMMAND catalog-entry-guard-test)
  endif()
  add_test(NAME catalog-cli-help COMMAND tyrell-catalog --help)
  add_test(NAME catalog-cli-version COMMAND tyrell-catalog --version)
  foreach(state fresh expired clock-unknown incompatible empty error)
    add_test(NAME catalog-ui-${state} COMMAND tyrell-catalog
      --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/${state}.ppm"
      --fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/${state}.tsv"
      --report "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/${state}.json" --frames 1)
  endforeach()
  add_test(NAME catalog-ui-details COMMAND tyrell-catalog
    --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/details.ppm"
    --fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fresh.tsv" --script down,enter
    --report "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/details.json" --frames 1)
  add_test(NAME catalog-ui-help COMMAND tyrell-catalog
    --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/help.ppm"
    --fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fresh.tsv" --script h
    --report "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/help.json" --frames 1)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fresh.tsv" CATALOG_ARTWORK_FIXTURE)
  file(SIZE "${CMAKE_CURRENT_SOURCE_DIR}/assets/tyrell-catalog.png" CATALOG_ICON_BYTES)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/artwork.tsv" "${CATALOG_ARTWORK_FIXTURE}ARTWORK\ttyrell-status@0.1.0-1\t${CATALOG_ICON_SHA}\t${CATALOG_ICON_BYTES}\t80\t80\t${CMAKE_CURRENT_SOURCE_DIR}/assets/tyrell-catalog.png\n")
  add_test(NAME catalog-ui-artwork COMMAND tyrell-catalog
    --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/artwork.ppm"
    --fixture "${CMAKE_CURRENT_BINARY_DIR}/artwork.tsv" --script enter
    --report "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/artwork.json" --frames 1)
  add_test(NAME catalog-ui-preview-no-transfer COMMAND tyrell-catalog
    --headless-screenshot "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/no-transfer.ppm"
    --fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fresh.tsv" --script r,d,a
    --report "${CMAKE_CURRENT_BINARY_DIR}/c1-previews/no-transfer.json" --frames 1)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/c1-previews")
endif()
install(TARGETS tyrell-catalog RUNTIME DESTINATION bin)
install(FILES assets/tyrell-catalog.desktop DESTINATION share/APPLaunch/applications)
install(FILES assets/tyrell-catalog-standard.desktop DESTINATION share/applications RENAME tyrell-catalog.desktop)
install(FILES assets/tyrell-catalog.png DESTINATION share/APPLaunch/share/images)
install(FILES assets/tyrell-catalog.png DESTINATION share/icons/hicolor/80x80/apps)
install(FILES "${LVGL_SOURCE_DIR}/LICENCE.txt" DESTINATION share/doc/tyrell-catalog RENAME lvgl-LICENCE.txt)
install(FILES "${TYRELL_APP_COMMON_SOURCE_DIR}/LICENSE" DESTINATION share/doc/tyrell-catalog RENAME app-common-LICENSE)
install(FILES backend.py catalog_cache.py catalog_metadata.py DESTINATION libexec/tyrell-catalog)
install(FILES config/trust.json config/source.json config/seed.json DESTINATION share/tyrell-catalog)
install(DIRECTORY config/artwork/ DESTINATION share/tyrell-catalog/artwork FILES_MATCHING PATTERN "*.png")
install(FILES README.md LICENSE THIRD_PARTY.md lodepng-LICENSE.txt DESTINATION share/doc/tyrell-catalog)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  execute_process(COMMAND dpkg --print-architecture OUTPUT_VARIABLE CATALOG_DEBIAN_ARCH
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT CATALOG_DEBIAN_ARCH STREQUAL "arm64")
    message(FATAL_ERROR "Catalog device package requires native Debian ARM64")
  endif()
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME tyrell-catalog)
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_CONTACT "Tyrell contributors <tyrell@localhost>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Offline signed discovery and verified downloads for Cardputer Zero")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
  set(CPACK_DEBIAN_PACKAGE_VERSION "${PROJECT_VERSION}-1")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
  set(CPACK_DEBIAN_PACKAGE_SECTION utils)
  set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.11), openssl (>= 3.0), ca-certificates")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
  set(CPACK_PACKAGE_FILE_NAME "tyrell-catalog_${PROJECT_VERSION}-1_arm64")
  include(CPack)
endif()
