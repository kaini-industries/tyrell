cmake_minimum_required(VERSION 3.25)
project(tyrell_app_common VERSION 0.1.0 LANGUAGES CXX)

add_library(tyrell_app_commands INTERFACE)
add_library(Tyrell::AppCommands ALIAS tyrell_app_commands)
target_include_directories(tyrell_app_commands INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(tyrell_app_commands INTERFACE cxx_std_17)

# Separate optional transport: command-only consumers introduce no IPC code.
add_library(tyrell_shell_link STATIC EXCLUDE_FROM_ALL src/shell_link.cpp)
add_library(Tyrell::ShellLink ALIAS tyrell_shell_link)
target_include_directories(tyrell_shell_link PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(tyrell_shell_link PUBLIC cxx_std_17)
target_compile_options(tyrell_shell_link PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
option(TYRELL_APP_COMMON_BUILD_TESTS "Build command contract tests" ${BUILD_TESTING})
if(TYRELL_APP_COMMON_BUILD_TESTS)
  add_executable(tyrell-app-commands-test tests/commands_test.cpp)
  target_link_libraries(tyrell-app-commands-test PRIVATE Tyrell::AppCommands)
  target_compile_options(tyrell-app-commands-test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME tyrell-app-commands COMMAND tyrell-app-commands-test)
endif()
option(TYRELL_APP_COMMON_BUILD_SHELL_LINK_TESTS "Build optional shell transport tests" OFF)
if(TYRELL_APP_COMMON_BUILD_SHELL_LINK_TESTS)
  add_executable(tyrell-shell-link-test tests/shell_link_test.cpp)
  target_link_libraries(tyrell-shell-link-test PRIVATE Tyrell::ShellLink)
  target_compile_options(tyrell-shell-link-test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME tyrell-shell-link COMMAND tyrell-shell-link-test)
endif()
