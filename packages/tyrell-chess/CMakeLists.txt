cmake_minimum_required(VERSION 3.25)
project(tyrell_chess VERSION 0.1.5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(CHESS_BACKEND HEADLESS CACHE STRING "HEADLESS, NATIVE or DESKTOP")
option(TYRELL_ENABLE_SHELL_LINK "Enable experimental native offline Chess shell lifecycle" OFF)
add_compile_definitions(TYRELL_ENABLE_SHELL_LINK=$<BOOL:${TYRELL_ENABLE_SHELL_LINK}>)
if(CHESS_BACKEND STREQUAL "NATIVE")
 add_compile_definitions(TYRELL_CHESS_NATIVE=1)
else()
 add_compile_definitions(TYRELL_CHESS_NATIVE=0)
endif()
if(NOT CHESS_BACKEND MATCHES "^(HEADLESS|NATIVE|DESKTOP)$")
 message(FATAL_ERROR "CHESS_BACKEND must be HEADLESS, NATIVE or DESKTOP")
endif()

set(TYRELL_GAME_BUILD_NETWORK OFF)
set(TYRELL_GAME_BUILD_EXAMPLES OFF)
set(TYRELL_GAME_BUILD_TESTS OFF)

if(NOT TARGET Tyrell::AppCommands)
 add_subdirectory(../tyrell-app-common tyrell-app-common)
endif()
if(TYRELL_ENABLE_SHELL_LINK)
 link_libraries(Tyrell::ShellLink)
endif()
add_library(chess_model src/game.cpp)
target_include_directories(chess_model PUBLIC src)
target_link_libraries(chess_model PUBLIC chess_engine Tyrell::AppCommands)
target_compile_options(chess_model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tyrell-chess src/main.cpp src/view.cpp src/lan.cpp)
target_link_libraries(tyrell-chess PRIVATE chess_model Tyrell::GameRuntime)
