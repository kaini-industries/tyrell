cmake_minimum_required(VERSION 3.25)
project(tyrell_chess VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(CHESS_BACKEND HEADLESS CACHE STRING "HEADLESS, NATIVE or DESKTOP")
if(NOT CHESS_BACKEND MATCHES "^(HEADLESS|NATIVE|DESKTOP)$")
 message(FATAL_ERROR "CHESS_BACKEND must be HEADLESS, NATIVE or DESKTOP")
endif()
set(TYRELL_GAME_BACKEND ${CHESS_BACKEND} CACHE STRING "" FORCE)
set(TYRELL_GAME_BUILD_NETWORK OFF)
set(TYRELL_GAME_BUILD_EXAMPLES OFF)
set(TYRELL_GAME_BUILD_TESTS OFF)
add_subdirectory(../tyrell-game tyrell-game)
add_subdirectory(engine)
add_library(chess_model src/game.cpp)
target_include_directories(chess_model PUBLIC src)
target_link_libraries(chess_model PUBLIC chess_engine)
target_compile_options(chess_model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tyrell-chess src/main.cpp src/view.cpp src/lan.cpp)
target_link_libraries(tyrell-chess PRIVATE chess_model Tyrell::GameRuntime)
target_compile_options(tyrell-chess PRIVATE -Wall -Wextra -Wpedantic)
if(CHESS_BACKEND STREQUAL "NATIVE")
 include(packaging.cmake)
endif()
