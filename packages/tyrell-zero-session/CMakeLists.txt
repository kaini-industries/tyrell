cmake_minimum_required(VERSION 3.25)
project(tyrell_zero_session VERSION 0.3.1 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build the package in the Linux ARM64 builder; pure state tests also run on the host.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX /usr CACHE PATH "Install prefix" FORCE)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(tyrell-zero-session-recovery src/recovery.cpp)
target_compile_options(tyrell-zero-session-recovery PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  target_compile_options(tyrell-zero-session-recovery PRIVATE -march=armv8-a -mtune=cortex-a53)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME session-state COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
  add_test(NAME recovery-version COMMAND tyrell-zero-session-recovery --version)
  add_test(NAME recovery-render COMMAND tyrell-zero-session-recovery --render-ppm "${CMAKE_CURRENT_BINARY_DIR}/recovery.ppm")
  add_test(NAME updates-help-text COMMAND tyrell-zero-session-recovery --updates --text)
  set_tests_properties(updates-help-text PROPERTIES PASS_REGULAR_EXPRESSION "reviewed base image")
  add_test(NAME updates-help-render COMMAND tyrell-zero-session-recovery --updates --render-ppm "${CMAKE_CURRENT_BINARY_DIR}/updates.ppm")
  add_test(NAME native-wizard-transaction COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/wizard/tests" -v)
endif()
install(TARGETS tyrell-zero-session-recovery RUNTIME DESTINATION libexec)
install(PROGRAMS bin/tyrell-zero-session-state bin/tyrell-zero-session DESTINATION libexec)
install(FILES src/state.py src/supervisor.py DESTINATION lib/tyrell-zero/session)
install(FILES wizard/completion.py DESTINATION lib/tyrell-zero/session RENAME wizard_complete.py)
install(PROGRAMS wizard/tyrell-zero-wizard-complete DESTINATION libexec)
set(TYRELL_NATIVE_WIZARD "" CACHE FILEPATH "Verified ARM64 native Wizard from the offline vendor build")
if(TYRELL_NATIVE_WIZARD)
  if(NOT EXISTS "${TYRELL_NATIVE_WIZARD}" OR IS_SYMLINK "${TYRELL_NATIVE_WIZARD}")
    message(FATAL_ERROR "A regular native Wizard binary is required")
  endif()
  install(PROGRAMS "${TYRELL_NATIVE_WIZARD}" DESTINATION libexec/tyrell-zero RENAME LaunchWizard)
endif()
install(FILES system/tyrell-zero-boot-state.service system/tyrell-zero-session.service system/tyrell-zero-wizard.service system/tyrell-zero.target DESTINATION lib/systemd/system)
install(FILES profile/logind.conf profile/activation.json DESTINATION share/tyrell-zero/profile)
install(FILES profile/integration-required.json DESTINATION share/tyrell-zero RENAME session-integration.json)
install(FILES README.md LICENSE DESTINATION share/doc/tyrell-zero-session)
execute_process(COMMAND dpkg --print-architecture OUTPUT_VARIABLE ARCH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ARCH STREQUAL "arm64")
  message(FATAL_ERROR "Device packages require native Debian ARM64")
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME tyrell-zero-session)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Tyrell contributors <tyrell@localhost>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Fail-closed Tyrell Zero native session and setup transaction")
set(CPACK_PACKAGE_DESCRIPTION "Boot/readiness validation, a normal-user PAM session supervisor, native Wizard completion and fixed recovery display. Activation requires separately verified offline integration and qualification. This files-only package does not activate a session.")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_VERSION "${PROJECT_VERSION}-1")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.11), greetd (= 0.10.3-4), libpam-systemd, dbus-user-session, systemd, applaunch (= 0.6.111)")
set(CPACK_DEBIAN_PACKAGE_SECTION admin)
set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGE_FILE_NAME "tyrell-zero-session_${PROJECT_VERSION}-1_arm64")
include(CPack)
