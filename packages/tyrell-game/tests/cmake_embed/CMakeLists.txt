cmake_minimum_required(VERSION 3.25)
project(tyrell_framework_consumer LANGUAGES C CXX)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Parent policy" FORCE)
set(BUILD_TESTING OFF)
set(TYRELL_GAME_BUILD_TESTS OFF)
set(TYRELL_GAME_BUILD_EXAMPLES OFF)
if(NOT DEFINED TYRELL_GAME_SOURCE)
  get_filename_component(TYRELL_GAME_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
add_subdirectory("${TYRELL_GAME_SOURCE}" framework)
if(NOT BUILD_SHARED_LIBS)
  message(FATAL_ERROR "Framework changed its parent's shared-library policy")
endif()
foreach(name tyrell_game_network tyrell_game_runtime tyrell_game_render)
  get_target_property(kind ${name} TYPE)
  if(NOT kind STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "${name} unexpectedly inherited a shared-library policy")
  endif()
endforeach()
add_executable(embedded-game-consumer main.cpp)
target_link_libraries(embedded-game-consumer PRIVATE Tyrell::GameRuntime Tyrell::GameRender Tyrell::GameNetwork)
