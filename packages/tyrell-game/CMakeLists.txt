# Test-only attachment shared by framework and consumer native lifecycle tests.
# This never changes production targets or opens real device nodes.
function(tyrell_game_attach_native_test_io target)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Native syscall fixtures require Linux")
  endif()
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tests/native_io_fixture.cpp")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tests")
  target_link_options(${target} PRIVATE
    -Wl,--wrap=open -Wl,--wrap=close -Wl,--wrap=read -Wl,--wrap=ioctl
    -Wl,--wrap=fstat -Wl,--wrap=mmap -Wl,--wrap=munmap -Wl,--wrap=fsync)
endfunction()

if(TYRELL_GAME_BUILD_NETWORK)
  if(NOT EXISTS "${ENET_SOURCE_DIR}/include/enet/enet.h")
    message(FATAL_ERROR "Set ENET_SOURCE_DIR from locks/game-build-inputs.json; CMake never downloads dependencies")
  endif()
  if(NOT TARGET enet)
    add_subdirectory("${ENET_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/enet" EXCLUDE_FROM_ALL)
    target_include_directories(enet PUBLIC "${ENET_SOURCE_DIR}/include")
  endif()
  add_library(tyrell_game_network STATIC src/net/transport.cpp)
  add_library(Tyrell::GameNetwork ALIAS tyrell_game_network)
  target_link_libraries(tyrell_game_network PUBLIC Tyrell::GameCore PRIVATE enet Threads::Threads)
  target_compile_options(tyrell_game_network PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(TYRELL_GAME_BUILD_RUNTIME)
  if(NOT EXISTS "${RAYLIB_SOURCE_DIR}/src/raylib.h")
    message(FATAL_ERROR "Set RAYLIB_SOURCE_DIR from locks/game-build-inputs.json; CMake never downloads dependencies")
  endif()
  if(NOT TYRELL_GAME_BACKEND MATCHES "^(DESKTOP|NATIVE|HEADLESS)$")
    message(FATAL_ERROR "TYRELL_GAME_BACKEND must be DESKTOP, NATIVE, or HEADLESS")
  endif()
  if(TYRELL_GAME_BACKEND STREQUAL "NATIVE" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native LCD backend requires Linux; HEADLESS also works on macOS")
  endif()
  if(TYRELL_GAME_BACKEND STREQUAL "DESKTOP")
    set(BUILD_EXAMPLES OFF)
    set(BUILD_SHARED_LIBS OFF)
    set(PLATFORM Desktop)
    add_subdirectory("${RAYLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/raylib" EXCLUDE_FROM_ALL)
    add_library(tyrell_game_graphics INTERFACE)
    target_link_libraries(tyrell_game_graphics INTERFACE raylib)
  else()
    add_library(tyrell_game_graphics STATIC
      "${RAYLIB_SOURCE_DIR}/src/rcore.c" "${RAYLIB_SOURCE_DIR}/src/rshapes.c"
      "${RAYLIB_SOURCE_DIR}/src/rtextures.c" "${RAYLIB_SOURCE_DIR}/src/rtext.c"
      "${RAYLIB_SOURCE_DIR}/src/rmodels.c")
    target_include_directories(tyrell_game_graphics PUBLIC "${RAYLIB_SOURCE_DIR}/src")
    set_target_properties(tyrell_game_graphics PROPERTIES C_STANDARD 11)
    target_compile_definitions(tyrell_game_graphics PRIVATE
      PLATFORM_MEMORY GRAPHICS_API_OPENGL_SOFTWARE SUPPORT_CUSTOM_FRAME_CONTROL=1
      SUPPORT_MODULE_RAUDIO=0 SUPPORT_SCREEN_CAPTURE=0 SUPPORT_AUTOMATION_EVENTS=0
      SUPPORT_CLIPBOARD_IMAGE=0 RLSW_FRAMEBUFFER_OUTPUT_BGRA=0)
    target_link_libraries(tyrell_game_graphics PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
  add_library(Tyrell::GameGraphics ALIAS tyrell_game_graphics)
  add_library(tyrell_game_render STATIC src/render.cpp)
  add_library(Tyrell::GameRender ALIAS tyrell_game_render)
  target_link_libraries(tyrell_game_render PUBLIC Tyrell::GameCore Tyrell::GameGraphics)
  target_compile_options(tyrell_game_render PRIVATE -Wall -Wextra -Wpedantic)
  add_library(tyrell_game_runtime STATIC src/runtime.cpp src/native.cpp src/input.cpp)
  add_library(Tyrell::GameRuntime ALIAS tyrell_game_runtime)
  target_link_libraries(tyrell_game_runtime PUBLIC Tyrell::GameCore Tyrell::GameGraphics PRIVATE Threads::Threads)
  target_compile_options(tyrell_game_runtime PRIVATE -Wall -Wextra -Wpedantic)
  if(NOT TYRELL_GAME_BACKEND STREQUAL "DESKTOP")
    target_compile_definitions(tyrell_game_runtime PRIVATE TYRELL_GAME_SOFTWARE=1)
    if(TYRELL_GAME_BACKEND STREQUAL "HEADLESS")
      target_compile_definitions(tyrell_game_runtime PRIVATE TYRELL_GAME_HEADLESS_DEFAULT=1)
    endif()
  endif()
endif()

      target_link_libraries(tyrell-beacon-arena PRIVATE Tyrell::ShellLink)
    endif()
