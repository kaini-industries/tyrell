cmake_minimum_required(VERSION 3.25)
project(tyrell_game VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(TYRELL_GAME_BUILD_RUNTIME "Build the raylib graphics runtime" ON)
option(TYRELL_GAME_BUILD_NETWORK "Build ENet transport" ON)
option(TYRELL_GAME_BUILD_TESTS "Build framework tests" ${BUILD_TESTING})
option(TYRELL_GAME_BUILD_EXAMPLES "Build the independent Beacon Arena example" ${PROJECT_IS_TOP_LEVEL})
set(TYRELL_GAME_BACKEND "DESKTOP" CACHE STRING "Graphics backend: DESKTOP, NATIVE, or HEADLESS")
set_property(CACHE TYRELL_GAME_BACKEND PROPERTY STRINGS DESKTOP NATIVE HEADLESS)
set(RAYLIB_SOURCE_DIR "" CACHE PATH "Verified raylib 6.2 source")
set(ENET_SOURCE_DIR "" CACHE PATH "Verified ENet 1.3.18 source")

add_library(tyrell_game_core INTERFACE)
add_library(Tyrell::GameCore ALIAS tyrell_game_core)
target_include_directories(tyrell_game_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(tyrell_game_core INTERFACE cxx_std_17)

if(TYRELL_GAME_BUILD_NETWORK OR TYRELL_GAME_BUILD_RUNTIME)
  find_package(Threads REQUIRED)
endif()
if(TYRELL_GAME_BUILD_NETWORK)
  if(NOT EXISTS "${ENET_SOURCE_DIR}/include/enet/enet.h")
    message(FATAL_ERROR "Set ENET_SOURCE_DIR from locks/game-build-inputs.json; CMake never downloads dependencies")
  endif()
  if(NOT TARGET enet)
    add_subdirectory("${ENET_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/enet" EXCLUDE_FROM_ALL)
    target_include_directories(enet PUBLIC "${ENET_SOURCE_DIR}/include")
  endif()
  add_library(tyrell_game_network STATIC src/net/transport.cpp)
  add_library(Tyrell::GameNetwork ALIAS tyrell_game_network)
  target_link_libraries(tyrell_game_network PUBLIC Tyrell::GameCore PRIVATE enet Threads::Threads)
  target_compile_options(tyrell_game_network PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(TYRELL_GAME_BUILD_RUNTIME)
  if(NOT EXISTS "${RAYLIB_SOURCE_DIR}/src/raylib.h")
    message(FATAL_ERROR "Set RAYLIB_SOURCE_DIR from locks/game-build-inputs.json; CMake never downloads dependencies")
  endif()
  if(NOT TYRELL_GAME_BACKEND MATCHES "^(DESKTOP|NATIVE|HEADLESS)$")
    message(FATAL_ERROR "TYRELL_GAME_BACKEND must be DESKTOP, NATIVE, or HEADLESS")
  endif()
  if(TYRELL_GAME_BACKEND STREQUAL "NATIVE" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native LCD backend requires Linux; HEADLESS also works on macOS")
  endif()
  if(TYRELL_GAME_BACKEND STREQUAL "DESKTOP")
    set(BUILD_EXAMPLES OFF)
    set(BUILD_SHARED_LIBS OFF)
    set(PLATFORM Desktop)
    add_subdirectory("${RAYLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/raylib" EXCLUDE_FROM_ALL)
    add_library(tyrell_game_graphics INTERFACE)
    target_link_libraries(tyrell_game_graphics INTERFACE raylib)
  else()
    add_library(tyrell_game_graphics STATIC
      "${RAYLIB_SOURCE_DIR}/src/rcore.c" "${RAYLIB_SOURCE_DIR}/src/rshapes.c"
      "${RAYLIB_SOURCE_DIR}/src/rtextures.c" "${RAYLIB_SOURCE_DIR}/src/rtext.c"
      "${RAYLIB_SOURCE_DIR}/src/rmodels.c")
    target_include_directories(tyrell_game_graphics PUBLIC "${RAYLIB_SOURCE_DIR}/src")
    set_target_properties(tyrell_game_graphics PROPERTIES C_STANDARD 11)
    target_compile_definitions(tyrell_game_graphics PRIVATE
      PLATFORM_MEMORY GRAPHICS_API_OPENGL_SOFTWARE SUPPORT_CUSTOM_FRAME_CONTROL=1
      SUPPORT_MODULE_RAUDIO=0 SUPPORT_SCREEN_CAPTURE=0 SUPPORT_AUTOMATION_EVENTS=0
      SUPPORT_CLIPBOARD_IMAGE=0 RLSW_FRAMEBUFFER_OUTPUT_BGRA=0)
    target_link_libraries(tyrell_game_graphics PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
  add_library(Tyrell::GameGraphics ALIAS tyrell_game_graphics)
  add_library(tyrell_game_render STATIC src/render.cpp)
  add_library(Tyrell::GameRender ALIAS tyrell_game_render)
  target_link_libraries(tyrell_game_render PUBLIC Tyrell::GameCore Tyrell::GameGraphics)
  target_compile_options(tyrell_game_render PRIVATE -Wall -Wextra -Wpedantic)
  add_library(tyrell_game_runtime STATIC src/runtime.cpp src/native.cpp src/input.cpp)
  add_library(Tyrell::GameRuntime ALIAS tyrell_game_runtime)
  target_link_libraries(tyrell_game_runtime PUBLIC Tyrell::GameCore Tyrell::GameGraphics PRIVATE Threads::Threads)
  target_compile_options(tyrell_game_runtime PRIVATE -Wall -Wextra -Wpedantic)
  if(NOT TYRELL_GAME_BACKEND STREQUAL "DESKTOP")
    target_compile_definitions(tyrell_game_runtime PRIVATE TYRELL_GAME_SOFTWARE=1)
    if(TYRELL_GAME_BACKEND STREQUAL "HEADLESS")
      target_compile_definitions(tyrell_game_runtime PRIVATE TYRELL_GAME_HEADLESS_DEFAULT=1)
    endif()
  endif()
endif()

if(TYRELL_GAME_BUILD_EXAMPLES AND TYRELL_GAME_BUILD_NETWORK)
  add_library(tyrell_beacon_session STATIC examples/beacon_session.cpp)
  target_include_directories(tyrell_beacon_session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/examples")
  target_link_libraries(tyrell_beacon_session PUBLIC Tyrell::GameNetwork Tyrell::GameCore)
  target_compile_options(tyrell_beacon_session PRIVATE -Wall -Wextra -Wpedantic)
  if(TYRELL_GAME_BUILD_RUNTIME)
    add_executable(tyrell-beacon-arena examples/beacon_arena.cpp)
    target_link_libraries(tyrell-beacon-arena PRIVATE tyrell_beacon_session Tyrell::GameRuntime Tyrell::GameRender)
    target_compile_options(tyrell-beacon-arena PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(TYRELL_GAME_BUILD_TESTS)
  enable_testing()
  add_executable(tyrell-game-core-test tests/core_test.cpp)
  target_link_libraries(tyrell-game-core-test PRIVATE Tyrell::GameCore)
  add_test(NAME tyrell-game-core COMMAND tyrell-game-core-test)
  if(TYRELL_GAME_BUILD_NETWORK)
    add_executable(tyrell-game-network-test tests/network_test.cpp)
    target_link_libraries(tyrell-game-network-test PRIVATE Tyrell::GameNetwork Threads::Threads)
    target_include_directories(tyrell-game-network-test PRIVATE "${ENET_SOURCE_DIR}/include")
    add_test(NAME tyrell-game-network COMMAND tyrell-game-network-test)
    set_tests_properties(tyrell-game-network PROPERTIES TIMEOUT 25)
  endif()
  if(TYRELL_GAME_BUILD_RUNTIME)
    add_executable(tyrell-game-input-test tests/input_test.cpp)
    target_link_libraries(tyrell-game-input-test PRIVATE Tyrell::GameRuntime)
    add_test(NAME tyrell-game-input COMMAND tyrell-game-input-test)
    if(NOT TYRELL_GAME_BACKEND STREQUAL "DESKTOP")
      add_executable(tyrell-game-runtime-test tests/runtime_test.cpp)
      target_link_libraries(tyrell-game-runtime-test PRIVATE Tyrell::GameRuntime)
      add_test(NAME tyrell-game-runtime COMMAND tyrell-game-runtime-test)
      set_tests_properties(tyrell-game-runtime PROPERTIES TIMEOUT 20)
      add_executable(tyrell-game-render-test tests/render_test.cpp)
      target_link_libraries(tyrell-game-render-test PRIVATE Tyrell::GameRuntime Tyrell::GameRender)
      add_test(NAME tyrell-game-render COMMAND tyrell-game-render-test)
      set_tests_properties(tyrell-game-render PROPERTIES TIMEOUT 15)
    endif()
  endif()
  if(TARGET tyrell_beacon_session)
    add_executable(tyrell-game-example-test tests/example_network_test.cpp)
    target_link_libraries(tyrell-game-example-test PRIVATE tyrell_beacon_session Threads::Threads)
    target_include_directories(tyrell-game-example-test PRIVATE "${ENET_SOURCE_DIR}/include")
    add_test(NAME tyrell-game-independent-example COMMAND tyrell-game-example-test)
    set_tests_properties(tyrell-game-independent-example PROPERTIES TIMEOUT 25)
  endif()
endif()
