cmake_minimum_required(VERSION 3.25)
project(tyrell_tanks VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TANKS_BUILD_DESKTOP "Build the windowed tank game" ON)
option(TANKS_BUILD_NATIVE "Build native LCD game and Debian package" OFF)
option(TANKS_BUILD_HEADLESS "Build the game with a display-free software renderer" OFF)
option(TANKS_BUILD_BENCHMARK "Build software and optional GLES display benchmarks" ON)
if((TANKS_BUILD_NATIVE AND TANKS_BUILD_DESKTOP) OR
   (TANKS_BUILD_HEADLESS AND TANKS_BUILD_DESKTOP) OR
   (TANKS_BUILD_HEADLESS AND TANKS_BUILD_NATIVE))
  message(FATAL_ERROR "Choose one of desktop, native, or headless game builds")
endif()
include(CTest)
set(TANKS_BUILD_GAME OFF)
if(TANKS_BUILD_DESKTOP OR TANKS_BUILD_NATIVE OR TANKS_BUILD_HEADLESS)
  set(TANKS_BUILD_GAME ON)
endif()
set(TYRELL_GAME_BUILD_RUNTIME ${TANKS_BUILD_GAME})
set(TYRELL_GAME_BUILD_NETWORK ON)
set(TYRELL_GAME_BUILD_EXAMPLES OFF)
if(TANKS_BUILD_NATIVE)
  set(TYRELL_GAME_BACKEND NATIVE)
elseif(TANKS_BUILD_HEADLESS)
  set(TYRELL_GAME_BACKEND HEADLESS)
else()
  set(TYRELL_GAME_BACKEND DESKTOP)
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../tyrell-game" "${CMAKE_CURRENT_BINARY_DIR}/tyrell-game")
add_library(tanks_world src/world.cpp)
target_include_directories(tanks_world PUBLIC src)
target_link_libraries(tanks_world PUBLIC Tyrell::GameCore)
target_compile_options(tanks_world PRIVATE -Wall -Wextra -Wpedantic)
add_library(tanks_network src/network.cpp)
target_link_libraries(tanks_network PUBLIC tanks_world Tyrell::GameNetwork)
target_compile_options(tanks_network PRIVATE -Wall -Wextra -Wpedantic)
if(TANKS_BUILD_GAME)
  add_executable(tyrell-tanks src/main.cpp src/scene.cpp)
  target_link_libraries(tyrell-tanks PRIVATE tanks_network Tyrell::GameRuntime Tyrell::GameRender)
  target_compile_options(tyrell-tanks PRIVATE -Wall -Wextra -Wpedantic)
  if(TANKS_BUILD_NATIVE OR TANKS_BUILD_HEADLESS)
    target_compile_definitions(tyrell-tanks PRIVATE TANKS_SOFTWARE=1)
  endif()
  if(TANKS_BUILD_NATIVE)
    include(packaging.cmake)
  endif()
endif()
if(TANKS_BUILD_BENCHMARK)
  add_subdirectory(benchmark)
endif()
if(BUILD_TESTING)
  add_executable(tanks-world-test tests/world_test.cpp)
  target_link_libraries(tanks-world-test PRIVATE tanks_world)
  add_test(NAME tanks-world COMMAND tanks-world-test)
  add_executable(tanks-network-test tests/network_test.cpp)
  target_link_libraries(tanks-network-test PRIVATE tanks_network)
  target_include_directories(tanks-network-test PRIVATE "${ENET_SOURCE_DIR}/include")
  add_test(NAME tanks-network COMMAND tanks-network-test)
  set_tests_properties(tanks-network PROPERTIES TIMEOUT 25)
  add_executable(tanks-peer-probe tests/peer_probe.cpp)
  target_link_libraries(tanks-peer-probe PRIVATE tanks_network)
  if(TARGET tyrell-tanks)
    add_test(NAME tanks-cli-help COMMAND tyrell-tanks --help)
    if(TANKS_BUILD_NATIVE OR TANKS_BUILD_HEADLESS)
      add_test(NAME tanks-framework-render COMMAND tyrell-tanks --headless --practice --autoplay
        --frames 5 --screenshot "${CMAKE_CURRENT_BINARY_DIR}/native-game.png"
        --report "${CMAKE_CURRENT_BINARY_DIR}/native-game.json")
      set_tests_properties(tanks-framework-render PROPERTIES TIMEOUT 15)
      find_package(Python3 COMPONENTS Interpreter QUIET)
      if(Python3_Interpreter_FOUND)
        add_test(NAME tanks-cli-session COMMAND "${Python3_EXECUTABLE}"
          "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py" --binary "$<TARGET_FILE:tyrell-tanks>")
        set_tests_properties(tanks-cli-session PROPERTIES TIMEOUT 20)
      endif()
    endif()
  endif()
endif()
