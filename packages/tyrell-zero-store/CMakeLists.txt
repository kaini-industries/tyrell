cmake_minimum_required(VERSION 3.25)
project(tyrell_zero_store VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX /usr CACHE PATH "Install prefix" FORCE)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(tyrell-zero-store-dispatcher src/dispatcher.cpp)
target_compile_options(tyrell-zero-store-dispatcher PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(store-routing-test tests/routing.cpp)
  target_include_directories(store-routing-test PRIVATE src)
  target_compile_options(store-routing-test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME store-routing COMMAND store-routing-test)
  add_test(NAME store-guard COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
endif()
install(PROGRAMS "$<TARGET_FILE:tyrell-zero-store-dispatcher>" DESTINATION share/APPLaunch/bin RENAME M5CardputerZero-AppStore)
install(PROGRAMS bin/appstore-package-helper DESTINATION libexec/tyrell-zero)
install(FILES guard.py DESTINATION lib/tyrell-zero/store)
install(FILES config/store-policy.json config/store-catalog.json DESTINATION share/tyrell-zero)
install(FILES README.md LICENSE DESTINATION share/doc/tyrell-zero-store)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  execute_process(COMMAND dpkg --print-architecture OUTPUT_VARIABLE ARCH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT ARCH STREQUAL "arm64")
    message(FATAL_ERROR "Device package requires native Debian ARM64")
  endif()
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME tyrell-zero-store)
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_CONTACT "Tyrell contributors <tyrell@localhost>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Qualified native M5 Store package boundary for Tyrell Zero")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
  set(CPACK_DEBIAN_PACKAGE_VERSION "${PROJECT_VERSION}-1")
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.11), dpkg, apt, applaunch (= 0.6.105)")
  set(CPACK_DEBIAN_PACKAGE_SECTION admin)
  set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_PACKAGE_FILE_NAME "tyrell-zero-store_${PROJECT_VERSION}-1_arm64")
  include(CPack)
endif()
